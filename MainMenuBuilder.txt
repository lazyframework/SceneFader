using System.IO;
using UnityEditor;
using UnityEditor.Events;
using UnityEditor.SceneManagement;
using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.UI;
using TMPro;

// Taruh di Assets/Editor
public class MainMenuBuilder : EditorWindow
{
    Sprite character;
    TMP_FontAsset customFont;
    string gameScene  = "GameScene";
    string playerName = "PLAYER";
    string agentTag   = "ETERAL WORLD";
    string rankText   = "ROOKIE 1";
    string version    = "v1.0.0  |  REGION: ID";
    float bottomY     = 60f;  // jarak baris tombol dari bawah layar
    float startRaise  = 0f;   // naikkan tombol START saja (0 = sejajar)
    Vector2 scrollPos;

    // --- PALET: MERAH + NAVY (tema tactical) ---
    static readonly Color Red        = new Color(0.87f, 0.18f, 0.26f);
    static readonly Color RedLine    = new Color(0.78f, 0.22f, 0.32f, 1f);
    static readonly Color RedDim     = new Color(0.55f, 0.17f, 0.25f, 1f);
    static readonly Color Navy       = new Color(0.07f, 0.10f, 0.22f, 0.93f);
    static readonly Color NavyDark   = new Color(0.04f, 0.06f, 0.15f, 0.92f);
    static readonly Color AccentBlue = new Color(0.45f, 0.62f, 1f);
    static readonly Color Gray       = new Color(0.62f, 0.67f, 0.78f);
    static readonly Color FrameCol   = new Color(0.78f, 0.2f, 0.3f, 0.5f);
    static readonly Vector2 Ctr      = new Vector2(0.5f, 0.5f);

    Sprite chamfer, chamferOutline;

    [MenuItem("Tools/Bangun Main Menu")]
    static void Open() => GetWindow<MainMenuBuilder>("Lobby Builder");

    void OnGUI()
    {
        GUILayout.Space(6);
        if (GUILayout.Button("BANGUN LOBBY", GUILayout.Height(40))) Build();
        GUILayout.Space(6);

        scrollPos = EditorGUILayout.BeginScrollView(scrollPos);
        GUILayout.Label("Aset", EditorStyles.boldLabel);
        character  = (Sprite)EditorGUILayout.ObjectField("Karakter (Opsional)", character, typeof(Sprite), false);
        customFont = (TMP_FontAsset)EditorGUILayout.ObjectField("Font TMP (Opsional)", customFont, typeof(TMP_FontAsset), false);

        GUILayout.Space(6);
        GUILayout.Label("Pengaturan", EditorStyles.boldLabel);
        gameScene  = EditorGUILayout.TextField("Nama Scene Game", gameScene);
        playerName = EditorGUILayout.TextField("Nama Player", playerName);
        agentTag   = EditorGUILayout.TextField("Tag (cth: AGENT 001)", agentTag);
        rankText   = EditorGUILayout.TextField("Rank", rankText);
        version    = EditorGUILayout.TextField("Teks versi", version);
        bottomY    = EditorGUILayout.FloatField("Jarak tombol dari bawah", bottomY);
        startRaise = EditorGUILayout.FloatField("START naik ke atas", startRaise);
        EditorGUILayout.EndScrollView();
    }

    void Build()
    {
        if (GameObject.Find("MenuManager") != null)
        {
            EditorUtility.DisplayDialog("Sudah ada", "Hapus Canvas dan MenuManager lama dulu.", "OK");
            return;
        }

        CreateSprites();

        // Canvas
        var canvasGO = new GameObject("Canvas", typeof(Canvas), typeof(CanvasScaler), typeof(GraphicRaycaster));
        canvasGO.GetComponent<Canvas>().renderMode = RenderMode.ScreenSpaceOverlay;
        var scaler = canvasGO.GetComponent<CanvasScaler>();
        scaler.uiScaleMode = CanvasScaler.ScaleMode.ScaleWithScreenSize;
        scaler.referenceResolution = new Vector2(1920, 1080);
        scaler.matchWidthOrHeight = 0.5f;
        Transform root = canvasGO.transform;

        if (Object.FindAnyObjectByType<EventSystem>() == null)
        {
            var es = new GameObject("EventSystem", typeof(EventSystem));
            es.AddComponent<StandaloneInputModule>();
        }

        // Karakter (opsional, di tengah)
        if (character != null)
        {
            var chr = NewImage("Karakter", root, character, Color.white);
            chr.preserveAspect = true;
            chr.raycastTarget = false;
            var cr = chr.rectTransform;
            cr.anchorMin = cr.anchorMax = cr.pivot = new Vector2(0.5f, 0f);
            cr.anchoredPosition = new Vector2(0, 180);
            cr.sizeDelta = new Vector2(1000, 800);
        }

        // Manager
        var menuGO = new GameObject("MenuManager");
        var menu = menuGO.AddComponent<MainMenu>();
        var lobbyUI = menuGO.AddComponent<LobbyUI>();

        // Garis bingkai atas & bawah
        FrameLine(root, true, 238f);
        FrameLine(root, false, 178f);

        // Profil (kiri atas, bisa diklik)
        var profileBtn = BuildProfile(root);

        // Status bar (kanan atas): SIGNAL, GOLD, MAIL, EVENTS
        Button mail, events;
        BuildStatusBar(root, lobbyUI, out mail, out events);

        // --- BARIS TOMBOL BAWAH (tengah) ---
        string[] names  = { "BtnShop", "BtnSettings", "BtnQuit", "BtnCharacter", "BtnInventory", "BtnStart" };
        string[] labels = { "SHOP", "SETTINGS", "QUIT", "CHARACTER", "INVENTORY", "START" };
        float w = 240f, h = 66f, gap = 20f;
        float total = labels.Length * w + (labels.Length - 1) * gap;
        var BC = new Vector2(0.5f, 0f);
        var row = new Button[labels.Length];
        for (int i = 0; i < row.Length; i++)
        {
            bool isStart = (i == row.Length - 1);
            float x = -total * 0.5f + w * 0.5f + i * (w + gap);
            float y = bottomY + (isStart ? startRaise : 0f);
            row[i] = MakeButton(root, names[i], labels[i], BC, BC, new Vector2(x, y),
                                new Vector2(w, h), isStart, 30);
        }
        var shop = row[0]; var settings = row[1]; var quit = row[2];
        var characters = row[3]; var inventory = row[4]; var start = row[5];

        // Versi (kiri bawah)
        var ver = NewText("Version", root, version, 22);
        ver.alignment = TextAlignmentOptions.Left;
        ver.color = new Color(Gray.r, Gray.g, Gray.b, 0.85f);
        Place(ver.rectTransform, Vector2.zero, Vector2.zero, new Vector2(46, 22), new Vector2(700, 30));

        // --- PANEL POP-UP ---
        var profilePanel    = BuildProfilePanel(root);
        var charactersPanel = BuildCharactersPanel(root, lobbyUI);
        var inventoryPanel  = BuildInventoryPanel(root);
        var shopPanel       = BuildShopPanel(root);
        var settingsPanel   = BuildSettingsPanel(root, lobbyUI);
        var mailPanel       = BuildMailPanel(root, lobbyUI);
        var eventsPanel     = BuildEventsPanel(root, lobbyUI);

        // Hubungkan tombol
        UnityEventTools.AddPersistentListener(start.onClick, menu.StartGame);
        UnityEventTools.AddPersistentListener(quit.onClick, menu.QuitGame);
        UnityEventTools.AddBoolPersistentListener(profileBtn.onClick, profilePanel.SetActive, true);
        UnityEventTools.AddBoolPersistentListener(characters.onClick, charactersPanel.SetActive, true);
        UnityEventTools.AddBoolPersistentListener(inventory.onClick, inventoryPanel.SetActive, true);
        UnityEventTools.AddBoolPersistentListener(shop.onClick, shopPanel.SetActive, true);
        UnityEventTools.AddBoolPersistentListener(settings.onClick, settingsPanel.SetActive, true);
        UnityEventTools.AddBoolPersistentListener(mail.onClick, mailPanel.SetActive, true);
        UnityEventTools.AddBoolPersistentListener(events.onClick, eventsPanel.SetActive, true);

        for (int i = 0; i < row.Length; i++) AddAnimation(row[i].gameObject, 0.1f + i * 0.06f);

        foreach (var p in new[] { profilePanel, charactersPanel, inventoryPanel, shopPanel, settingsPanel, mailPanel, eventsPanel })
            p.SetActive(false);

        EditorUtility.SetDirty(lobbyUI);

        var so = new SerializedObject(menu);
        so.FindProperty("gameSceneName").stringValue = gameScene;
        so.ApplyModifiedProperties();

        EditorSceneManager.MarkSceneDirty(EditorSceneManager.GetActiveScene());
        Selection.activeGameObject = canvasGO;
    }

    // =========================== HUD UTAMA ===========================

    // Garis tipis merah + ujung tegak di kiri & kanan
    void FrameLine(Transform root, bool top, float offset)
    {
        float ay = top ? 1f : 0f;
        float y = top ? -offset : offset;

        var line = NewImage(top ? "FrameTop" : "FrameBottom", root, null, FrameCol);
        line.raycastTarget = false;
        var r = line.rectTransform;
        r.anchorMin = new Vector2(0, ay);
        r.anchorMax = new Vector2(1, ay);
        r.pivot = Ctr;
        r.anchoredPosition = new Vector2(0, y);
        r.sizeDelta = new Vector2(-140, 2);

        for (int i = 0; i < 2; i++)
        {
            var tick = NewImage("Tick", root, null, FrameCol);
            tick.raycastTarget = false;
            var a = new Vector2(i == 0 ? 0f : 1f, ay);
            Place(tick.rectTransform, a, Ctr, new Vector2(i == 0 ? 70 : -70, y), new Vector2(2, 26));
        }
    }

    // Kotak profil kiri atas (bisa diklik)
    Button BuildProfile(Transform root)
    {
        var TL = new Vector2(0, 1);
        var box = NewImage("Profile", root, chamfer, Navy);
        box.type = Image.Type.Sliced;
        box.raycastTarget = true;
        Place(box.rectTransform, TL, TL, new Vector2(62, -55), new Vector2(440, 145));

        var outline = NewImage("Outline", box.transform, chamferOutline, RedLine);
        outline.type = Image.Type.Sliced;
        outline.raycastTarget = false;
        Stretch(outline.rectTransform);

        Bracket(box.transform, true);
        Bracket(box.transform, false);

        var avatar = NewImage("Avatar", box.transform, null, new Color(0.10f, 0.14f, 0.28f));
        avatar.raycastTarget = false;
        Place(avatar.rectTransform, TL, TL, new Vector2(16, -16), new Vector2(92, 92));
        RectOutline(avatar.transform, RedLine, 2f);

        var n = NewText("Name", box.transform, playerName + "  |  " + agentTag, 26);
        n.fontStyle = FontStyles.Bold;
        n.alignment = TextAlignmentOptions.Left;
        n.enableWordWrapping = false;
        Place(n.rectTransform, TL, new Vector2(0, 0.5f), new Vector2(128, -34), new Vector2(310, 34));

        var rank = NewText("Rank", box.transform, rankText, 26);
        rank.fontStyle = FontStyles.Bold;
        rank.alignment = TextAlignmentOptions.Left;
        rank.color = new Color(1f, 0.32f, 0.38f);
        Place(rank.rectTransform, TL, new Vector2(0, 0.5f), new Vector2(128, -70), new Vector2(310, 32));

        var peak = NewText("Peak", box.transform, "PEAK: " + rankText, 22);
        peak.fontStyle = FontStyles.Bold;
        peak.alignment = TextAlignmentOptions.Left;
        peak.color = AccentBlue;
        Place(peak.rectTransform, TL, new Vector2(0, 0.5f), new Vector2(128, -100), new Vector2(310, 30));

        var badge = NewImage("LevelBadge", box.transform, null, NavyDark);
        badge.raycastTarget = false;
        Place(badge.rectTransform, Vector2.zero, Vector2.zero, new Vector2(16, 10), new Vector2(64, 24));
        RectOutline(badge.transform, Gray, 1f);
        var lv = NewText("Text", badge.transform, "LVL 1", 17);
        lv.fontStyle = FontStyles.Bold;
        Stretch(lv.rectTransform);

        var btn = box.gameObject.AddComponent<Button>();
        btn.targetGraphic = box;
        btn.transition = Selectable.Transition.ColorTint;
        var c = btn.colors;
        c.normalColor = new Color(0.9f, 0.9f, 0.95f);
        c.highlightedColor = Color.white;
        c.pressedColor = new Color(0.65f, 0.65f, 0.7f);
        c.selectedColor = c.normalColor;
        c.fadeDuration = 0.1f;
        btn.colors = c;
        return btn;
    }

    // Tanda kurung di sudut kotak (kiri-atas / kanan-atas)
    void Bracket(Transform parent, bool left)
    {
        var col = new Color(0.85f, 0.88f, 0.95f, 0.8f);
        var a = left ? new Vector2(0, 1) : new Vector2(1, 1);
        float sx = left ? -8f : 8f;

        var hz = NewImage("BracketH", parent, null, col);
        hz.raycastTarget = false;
        Place(hz.rectTransform, a, a, new Vector2(sx, 8), new Vector2(30, 2));

        var vt = NewImage("BracketV", parent, null, col);
        vt.raycastTarget = false;
        Place(vt.rectTransform, a, a, new Vector2(sx, 8), new Vector2(2, 30));
    }

    // Status kanan atas: SIGNAL, GOLD, MAIL, EVENTS
    void BuildStatusBar(Transform root, LobbyUI lobbyUI, out Button mail, out Button events)
    {
        var TR = new Vector2(1, 1);
        float y = -55f, h = 46f;

        events = MakeFlatButton(root, "BtnEvents", "EVENTS", TR, TR, new Vector2(-66, y), new Vector2(120, h), 22);
        mail   = MakeFlatButton(root, "BtnMail", "MAIL", TR, TR, new Vector2(-194, y), new Vector2(100, h), 22);

        // GOLD
        var goldBox = FlatBox(root, "GoldBox", TR, new Vector2(-302, y), new Vector2(190, h));
        var coin = NewImage("Coin", goldBox.transform, null, new Color(1f, 0.78f, 0.2f));
        coin.raycastTarget = false;
        Place(coin.rectTransform, new Vector2(0, 0.5f), Ctr, new Vector2(24, 0), new Vector2(16, 16));
        coin.rectTransform.localRotation = Quaternion.Euler(0, 0, 45);

        var gl = NewText("GoldLabel", goldBox.transform, "GOLD", 19);
        gl.fontStyle = FontStyles.Bold;
        gl.alignment = TextAlignmentOptions.Left;
        gl.color = Gray;
        Place(gl.rectTransform, new Vector2(0, 0.5f), new Vector2(0, 0.5f), new Vector2(42, 0), new Vector2(70, 30));

        var gv = NewText("GoldValue", goldBox.transform, "0", 26);
        gv.fontStyle = FontStyles.Bold;
        gv.alignment = TextAlignmentOptions.Right;
        Place(gv.rectTransform, new Vector2(1, 0.5f), new Vector2(1, 0.5f), new Vector2(-14, 0), new Vector2(100, 34));
        lobbyUI.goldText = gv;

        // SIGNAL (hiasan)
        var sig = FlatBox(root, "SignalBox", TR, new Vector2(-500, y), new Vector2(184, h));
        for (int k = 0; k < 4; k++)
        {
            var bar = NewImage("Bar" + k, sig.transform, null, new Color(0.75f, 0.8f, 0.9f));
            bar.raycastTarget = false;
            Place(bar.rectTransform, Vector2.zero, Vector2.zero, new Vector2(16 + 9 * k, 12), new Vector2(6, 6 + 4 * k));
        }
        var st = NewText("SignalText", sig.transform, "ONLINE", 19);
        st.fontStyle = FontStyles.Bold;
        st.alignment = TextAlignmentOptions.Left;
        st.color = Gray;
        Place(st.rectTransform, new Vector2(0, 0.5f), new Vector2(0, 0.5f), new Vector2(60, 0), new Vector2(110, 30));
    }

    GameObject BuildProfilePanel(Transform root)
    {
        var panel = MakePanel(root, "ProfilePanel", "PROFILE");
        var win = panel.transform.Find("Window");

        Label(win, playerName + "   |   " + agentTag, 30, new Vector2(0, 165), new Vector2(700, 44),

        string[] names  = { "LEVEL", "HP", "MP", "ATK", "DEF", "SPD" };
        string[] values = { "1", "100 / 100", "100 / 100", "10", "5", "5" };
        for (int i = 0; i < names.Length; i++)
        {
            float x = (i % 2 == 0) ? -205 : 205;

            Label(r.transform, names[i], 28, new Vector2(-90, 0), new Vector2(160, 40),

            Label(r.transform, values[i], 28, new Vector2(90, 0), new Vector2(170, 40),
                  TextAlignmentOptions.Right, new Color(1f, 0.4f, 0.45f), true);
        }

    }

    {

        var win = panel.transform.Find("Window");

        string[] names = { "ARION", "LUNA", "KAI", "ZERO" };
        Color[] tints = {
            new Color(0.2f, 0.5f, 0.95f), new Color(0.6f, 0.35f, 0.9f),
            new Color(0.15f, 0.7f, 0.65f), new Color(0.85f, 0.4f, 0.5f)
        };
        float w = 190f, gap = 14f;

        for (int i = 0; i < names.Length; i++)

            card.raycastTarget = true;

            var b = card.gameObject.AddComponent<Button>();
            b.targetGraphic = card;
            b.transition = Selectable.Transition.ColorTint;
            var c = b.colors;
            c.normalColor = new Color(0.9f, 0.9f, 0.95f);
            c.highlightedColor = Color.white;
            c.pressedColor = new Color(0.65f, 0.65f, 0.7f);
            c.selectedColor = c.normalColor;
            c.fadeDuration = 0.1f;
            b.colors = c;

            Box(card.transform, "Portrait", new Vector2(0, 35), new Vector2(150, 140), tints[i]);
            Label(card.transform, names[i], 30, new Vector2(0, -75), new Vector2(170, 40),
                  TextAlignmentOptions.Center, Color.white, true);

            UnityEventTools.AddIntPersistentListener(b.onClick, lobbyUI.SelectCharacter, i);

        }
        lobbyUI.characterCards = cards;
        return panel;
    }

    // =========================== KOMPONEN ===========================

    // Tombol bersudut terpotong (kiri-atas & kanan-bawah). Primary = merah solid (START)
    Button MakeButton(Transform parent, string name, string label, Vector2 anchor, Vector2 pivot,
                      Vector2 pos, Vector2 size, bool isPrimary, float fontSize = 30)
    {
        var go = new GameObject(name, typeof(RectTransform), typeof(Image));
        go.transform.SetParent(parent, false);
        Place(go.GetComponent<RectTransform>(), anchor, pivot, pos, size);

        var bg = go.GetComponent<Image>();
        bg.sprite = chamfer;
        bg.type = Image.Type.Sliced;
        bg.color = isPrimary ? Red : Navy;

        var border = NewImage("Outline", go.transform, chamferOutline,
            isPrimary ? new Color(1f, 0.55f, 0.6f, 0.9f) : RedLine);
        border.type = Image.Type.Sliced;
        border.raycastTarget = false;
        Stretch(border.rectTransform);

        var txt = NewText("Label", go.transform, label, fontSize);
        txt.fontStyle = FontStyles.Bold;
        txt.characterSpacing = 2;
        txt.color = Color.white;
        Stretch(txt.rectTransform);

        var btn = go.AddComponent<Button>();
        btn.targetGraphic = bg;
        btn.transition = Selectable.Transition.ColorTint;
        var c = btn.colors;
        c.normalColor      = new Color(0.88f, 0.88f, 0.92f);
        c.highlightedColor = Color.white;
        c.pressedColor     = new Color(0.62f, 0.62f, 0.68f);
        c.selectedColor    = c.normalColor;
        c.fadeDuration     = 0.1f;
        btn.colors = c;
        return btn;
    }

    // Tombol persegi datar kecil (MAIL / EVENTS)
    Button MakeFlatButton(Transform parent, string name, string label, Vector2 anchor, Vector2 pivot,
                          Vector2 pos, Vector2 size, float fontSize)
    {
        var go = new GameObject(name, typeof(RectTransform), typeof(Image));
        go.transform.SetParent(parent, false);
        Place(go.GetComponent<RectTransform>(), anchor, pivot, pos, size);

    // --- PEMBUAT SPRITE (persegi panjang, sudut kiri-atas & kanan-bawah terpotong) ---
    delegate bool InsideFn(float x, float y, float w, float h);

    void CreateSprites()
    {
        Directory.CreateDirectory(Application.dataPath + "/MenuSprites");

        chamfer = Shape("menu_chamfer", 256, 64, new Vector4(22, 22, 22, 22),
            (x, y, w, h) => Chamfer(x, y, w, h, 18f, 0f));
        chamferOutline = Shape("menu_chamfer_outline", 256, 64, new Vector4(22, 22, 22, 22),
            (x, y, w, h) => Chamfer(x, y, w, h, 18f, 0f) && !Chamfer(x, y, w, h, 18f, 2.5f));
    }

    static bool Chamfer(float x, float y, float w, float h, float c, float inset)
    {
        if (x < inset || x > w - inset || y < inset || y > h - inset) return false;
        float d = inset * 1.4142f;
        if (x + (h - y) < c + d) return false;   // potong kiri-atas
        if ((w - x) + y < c + d) return false;   // potong kanan-bawah
        return true;
    }

    Sprite Shape(string name, int w, int h, Vector4 border, InsideFn inside)
    {
        const int SS = 3;
        var tex = new Texture2D(w, h, TextureFormat.RGBA32, false);
        for (int y = 0; y < h; y++)
            for (int x = 0; x < w; x++)
            {
                int hit = 0;
                for (int sy = 0; sy < SS; sy++)
                    for (int sx = 0; sx < SS; sx++)
                        if (inside(x + (sx + 0.5f) / SS, y + (sy + 0.5f) / SS, w, h)) hit++;
                tex.SetPixel(x, y, new Color(1, 1, 1, hit / (float)(SS * SS)));
            }

        string path = $"Assets/MenuSprites/{name}.png";
        File.WriteAllBytes(Application.dataPath + $"/MenuSprites/{name}.png", tex.EncodeToPNG());
        DestroyImmediate(tex);
        AssetDatabase.ImportAsset(path, ImportAssetOptions.ForceUpdate);

        var imp = (TextureImporter)AssetImporter.GetAtPath(path);
        imp.textureType = TextureImporterType.Sprite;
        imp.spriteImportMode = SpriteImportMode.Single;
        imp.alphaIsTransparency = true;
        imp.mipmapEnabled = false;
        imp.spriteBorder = border;
        imp.SaveAndReimport();
        return AssetDatabase.LoadAssetAtPath<Sprite>(path);
    }

    // --- HELPER ---
    void Place(RectTransform r, Vector2 anchor, Vector2 pivot, Vector2 pos, Vector2 size)
    {
        r.anchorMin = r.anchorMax = anchor;
        r.pivot = pivot;
        r.anchoredPosition = pos;
        r.sizeDelta = size;
    }

    Image NewImage(string name, Transform parent, Sprite s, Color c)
    {
        var go = new GameObject(name, typeof(RectTransform), typeof(Image));
        go.transform.SetParent(parent, false);
        var img = go.GetComponent<Image>();
        img.sprite = s;
        img.color = c;
        return img;
    }

    TextMeshProUGUI NewText(string name, Transform parent, string text, float size)
    {
        var go = new GameObject(name, typeof(RectTransform), typeof(TextMeshProUGUI));
        go.transform.SetParent(parent, false);
        var t = go.GetComponent<TextMeshProUGUI>();
        t.text = text;
        t.fontSize = size;
        t.alignment = TextAlignmentOptions.Center;
        t.color = Color.white;
        t.raycastTarget = false;
        if (customFont != null) t.font = customFont;
        return t;
    }

    void Stretch(RectTransform r)
    {
        r.anchorMin = Vector2.zero;
        r.anchorMax = Vector2.one;
        r.offsetMin = r.offsetMax = Vector2.zero;
    }

    void AddAnimation(GameObject btnObj, float delay)
    {
        var anim = btnObj.AddComponent<MenuAnimator>();
        anim.delay = delay;
    }
